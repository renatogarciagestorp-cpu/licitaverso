    function esconderSumulasEAnalisesEEnunciadosRelacionados() {
        var divAguardeDeSumulas = $("#divAguarde");
        var divSumulas = $("#divSumulas");
        var divAnalises = $("#divAnalises");
        var divEnunciadosRelacionados = $("#divEnunciadosRelacionados");

        divAguardeDeSumulas.show();
        divSumulas.hide();
        divAnalises.hide();
        divEnunciadosRelacionados.hide();
    }

    function mostrarSumulasEAnalisesEEnunciadosRelacionados() {
        var divAguardeDeSumulas = $("#divAguarde");
        var divSumulas = $("#divSumulas");
        var divAnalises = $("#divAnalises");
        var divEnunciadosRelacionados = $("#divEnunciadosRelacionados");

        divAguardeDeSumulas.hide();
        divAnalises.show();
        divSumulas.show();
        divEnunciadosRelacionados.show();
    }

    function limpaDivSumulasEAnalisesEEnunciadosRelacionados(){
        var divSumulas = $("#divSumulas");
        var divAnalises = $("#divAnalises");
        var divEnunciadosRelacionados = $("#divEnunciadosRelacionados");

        divSumulas.empty();
        divAnalises.empty();
        divEnunciadosRelacionados.empty();
    }

    function carregarTooltips() {
        $("[data-toggle='tooltip']").tooltip();
    }

    $(document).ready(function() {
        $("input[name='optTema']").change(function() {
            limpaDivSumulasEAnalisesEEnunciadosRelacionados();

            esconderSumulasEAnalisesEEnunciadosRelacionados();

            var enunciadoID = $(this).val();

            var millisecondsToWait = 500;
            setTimeout(function() {
                $.ajax({
                    url: "/tcmjuris/Enunciados/SumulasPorEnunciado",
                    type: "POST",
                    dataType: "html",
                    cache: false,
                    async: false,
                    data: { enunciadoID: enunciadoID }
                }).done(function(data) {
                    var divSumulas = $("#divSumulas");
                    divSumulas.empty().append(data);
                }).fail(function(jqXHR, textStatus, errorThrown) {
                    Tcmrj.helper.trataRetornoDeErroDeAjax(jqXHR, textStatus, errorThrown);
                });

                $.ajax({
                    url: "/tcmjuris/Enunciados/EnunciadosRelacionadosPorEnunciado",
                    type: "POST",
                    dataType: "html",
                    cache: false,
                    async: false,
                    data: { enunciadoID: enunciadoID }
                }).done(function(data) {
                    var divEnunciadosRelacionados = $("#divEnunciadosRelacionados");
                    divEnunciadosRelacionados.empty().append(data);
                }).fail(function(jqXHR, textStatus, errorThrown) {
                    Tcmrj.helper.trataRetornoDeErroDeAjax(jqXHR, textStatus, errorThrown);
                });

                $.ajax({
                    url: "/tcmjuris/Enunciados/AnalisesPorEnunciado",
                    type: "POST",
                    dataType: "html",
                    cache: false,
                    async: false,
                    data: { enunciadoID: enunciadoID }
                }).done(function(data) {
                    var divAnalises = $("#divAnalises");
                    divAnalises.empty().append(data);
                    carregarTooltips();
                }).fail(function(jqXHR, textStatus, errorThrown) {
                    Tcmrj.helper.trataRetornoDeErroDeAjax(jqXHR, textStatus, errorThrown);
                });

                var divSumulas = $("#divSumulas");
                mostrarSumulasEAnalisesEEnunciadosRelacionados();
                $('html, body').animate({
                    scrollTop: divSumulas.offset().top - 100
            }, 1000);
            }, millisecondsToWait);


        });
    });
